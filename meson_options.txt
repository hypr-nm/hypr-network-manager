option(
  'install-polkit',
  type: 'boolean',
  value: true,
  description: 'Install polkit policy/rules so wheel/sudo users can run the vendored create_ap hotspot helper without a pkexec password prompt on every invocation. Disable for distros that ship their own polkit rules or for unprivileged build sandboxes.'
)

option(
  'install-dev-polkit',
  type: 'boolean',
  value: false,
  description: 'Install the dev-tree create_ap polkit action whose exec.path points at the (user-writable) source tree. SECURITY: this is off by default and never grants passwordless execution; the action always requires auth_admin because the target script is not root-owned. Enable only for developer machines that run uninstalled builds and accept a password prompt per hotspot start/stop.'
)
